To access the frontend service, run:

  export POD_NAME=$(kubectl get pods --namespace {{ .Release.Namespace }} -l "app.kubernetes.io/name={{ include "todo-app.name" . }},app.kubernetes.io/instance={{ .Release.Name }},app.kubernetes.io/component=frontend" -o jsonpath="{.items[0].metadata.name}")

  kubectl port-forward --namespace {{ .Release.Namespace }} svc/{{ include "todo-app.fullname" . }}-backend 8000:8000

  kubectl get pods -n {{ .Release.Namespace }}
